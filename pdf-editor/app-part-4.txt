function cssColorToHex(c){if(/^#[0-9a-f]{6}$/i.test(c))return c.toLowerCase();const m=c.match(/rgba?\((\d+)\D+(\d+)\D+(\d+)/i);if(!m)return'#000000';return'#'+[m[1],m[2],m[3]].map(v=>(+v).toString(16).padStart(2,'0')).join('')}
