function inferFont(page,it,styles){const style=styles?.[it.fontName]||{};let obj=null;try{obj=page.commonObjs.get(it.fontName)}catch{}const label=[style.fontFamily,obj?.name,obj?.fallbackName,obj?.loadedName].filter(Boolean).join(' ');let family='Helvetica';if(/courier|mono/i.test(label))family='Courier';else if(/times|serif|roman/i.test(label))family='Times';const bold=/bold|black|heavy|semibold|semi-bold|demi/i.test(label),italic=/italic|oblique/i.test(label);return{family,bold,italic}}
async function extractPdfText(page){try{const tc=await page.getTextContent(),raw=[],W=viewport.width,H=viewport.height;for(const it of tc.items||[]){if(!it?.str?.trim()||!it.transform)continue;const tx=pdfjsLib.Util.transform(viewport.transform,it.transform),angle=Math.atan2(tx[1],tx[0]);if(Math.abs(Math.sin(angle))>.12)continue;const cssH=Math.max(4,Math.hypot(tx[2],tx[3])),cssW=Math.max(2,Math.abs((it.width||0)*viewport.scale)),size=Math.max(4,Math.hypot(it.transform[2],it.transform[3])||it.height||12),xPx=tx[4],baseline=tx[5],top=baseline-cssH,sidePad=Math.max(1.25,cssH*.08),mx=Math.max(0,xPx-sidePad),my=Math.max(0,baseline-cssH*1.15),mr=Math.min(W,xPx+cssW+sidePad),mb=Math.min(H,baseline+cssH*.40),sourceKey=[it.str,Number(it.transform[4]||0).toFixed(3),Number(it.transform[5]||0).toFixed(3),Number(it.width||0).toFixed(3),Number(size||0).toFixed(3)].join('|'),f=inferFont(page,it,tc.styles);raw.push({text:it.str,xPx,baseline,top,cssW,cssH,size,sourceKey,maskRect:{x:mx/W,y:my/H,w:(mr-mx)/W,h:(mb-my)/H},style:{family:f.family,size,bold:f.bold,italic:f.italic,underline:false,strike:false,script:'normal',color:'#000000'}})}pdfTextLines=groupLines(raw,W,H)}catch(e){console.warn(e);pdfTextLines=[]}}
function groupLines(items,W,H){const sorted=[...items].sort((a,b)=>a.baseline-b.baseline||a.xPx-b.xPx),lines=[];for(const it of sorted){let line=lines.find(l=>Math.abs(l.baseline-it.baseline)<=Math.max(2.5,Math.max(l.cssH,it.cssH)*.32));if(!line){line={baseline:it.baseline,cssH:it.cssH,items:[]};lines.push(line)}line.items.push(it);line.cssH=Math.max(line.cssH,it.cssH)}const groups=[];for(const line of lines){line.items.sort((a,b)=>a.xPx-b.xPx);let bucket=[];const flush=()=>{if(!bucket.length)return;const first=bucket[0],runs=[],masks=[],keys=[];let prevEnd=first.xPx;for(const item of bucket){const gap=item.xPx-prevEnd;if(runs.length&&gap>Math.max(1,item.cssH*.18)){const spaces=Math.max(1,Math.round(gap/Math.max(3,item.size*.28)));runs.push({text:' '.repeat(Math.min(spaces,8)),...cloneStyle(item.style)})}runs.push({text:item.text,...cloneStyle(item.style)});masks.push({sourceKey:item.sourceKey,rect:structuredClone(item.maskRect),pdfRect:maskPdfRect(item.maskRect),maskColor:sampleMaskColor(item.maskRect)});keys.push(item.sourceKey);prevEnd=item.xPx+item.cssW}const left=first.xPx,right=Math.max(...bucket.map(i=>i.xPx+i.cssW)),top=Math.min(...bucket.map(i=>i.top));groups.push({x:left/W,y:top/H,w:(right-left)/W,h:Math.max(...bucket.map(i=>i.cssH))*1.3/H,runs:normalizeRuns(runs),sourceMasks:masks,sourceKeys:keys,align:'left'});bucket=[]};for(const it of line.items){if(bucket.length){const prev=bucket[bucket.length-1],gap=it.xPx-(prev.xPx+prev.cssW);if(gap>Math.max(28,line.cssH*3.2))flush()}bucket.push(it)}flush()}return groups}
function maskPdfRect(r){const x1=r.x*viewport.width,y1=r.y*viewport.height,x2=(r.x+r.w)*viewport.width,y2=(r.y+r.h)*viewport.height,pts=[viewport.convertToPdfPoint(x1,y1),viewport.convertToPdfPoint(x2,y1),viewport.convertToPdfPoint(x1,y2),viewport.convertToPdfPoint(x2,y2)],xs=pts.map(p=>p[0]),ys=pts.map(p=>p[1]);return{x:Math.min(...xs),y:Math.min(...ys),w:Math.max(...xs)-Math.min(...xs),h:Math.max(...ys)-Math.min(...ys)}}
