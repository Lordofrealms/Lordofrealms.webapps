async function extractPdfText(page){
try{
const tc=await page.getTextContent(),items=[],W=viewport.width,H=viewport.height;
for(const it of tc.items||[]){
if(!it?.str?.trim()||!it.transform)continue;
const tx=pdfjsLib.Util.transform(viewport.transform,it.transform),angle=Math.atan2(tx[1],tx[0]);
if(Math.abs(Math.sin(angle))>.12)continue;
const cssH=Math.max(4,Math.hypot(tx[2],tx[3]));
const cssW=Math.max(3,Math.abs((it.width||0)*viewport.scale));
const size=Math.max(4,Math.hypot(it.transform[2],it.transform[3])||it.height||12);
const textX=tx[4], baselineY=tx[5];
const sidePad=Math.max(1.25,cssH*.08);
const maskTop=baselineY-cssH*1.15;
const maskBottom=baselineY+cssH*.40;
const mx=Math.max(0,textX-sidePad), my=Math.max(0,maskTop);
const mr=Math.min(W,textX+cssW+sidePad), mb=Math.min(H,maskBottom);
const sourceKey=[it.str,Number(it.transform[4]||0).toFixed(3),Number(it.transform[5]||0).toFixed(3),Number(it.width||0).toFixed(3),Number(size||0).toFixed(3)].join('|');
items.push({
text:it.str,
x:textX/W,
y:(baselineY-cssH)/H,
w:cssW/W,
h:Math.max(cssH*1.18/H,.006),
size,
sourceKey,
maskRect:{x:mx/W,y:my/H,w:Math.max(0,(mr-mx)/W),h:Math.max(0,(mb-my)/H)}
});
}
pdfTextItems=items;
}catch(e){console.warn(e);pdfTextItems=[]}
}
function maskPdfRect(r){
const x1=r.x*viewport.width,y1=r.y*viewport.height,x2=(r.x+r.w)*viewport.width,y2=(r.y+r.h)*viewport.height;
const pts=[viewport.convertToPdfPoint(x1,y1),viewport.convertToPdfPoint(x2,y1),viewport.convertToPdfPoint(x1,y2),viewport.convertToPdfPoint(x2,y2)];
const xs=pts.map(p=>p[0]),ys=pts.map(p=>p[1]);
return{x:Math.min(...xs),y:Math.min(...ys),w:Math.max(...xs)-Math.min(...xs),h:Math.max(...ys)-Math.min(...ys)};
}
function sampleMaskColor(r){
try{
const sx=pdfCanvas.width/viewport.width,sy=pdfCanvas.height/viewport.height,W=viewport.width,H=viewport.height;
const left=r.x*W,right=(r.x+r.w)*W,top=r.y*H,bottom=(r.y+r.h)*H,pad=Math.max(3,Math.min(8,(bottom-top)*.3));
const pts=[
[left-pad,top-pad],[(left+right)/2,top-pad],[right+pad,top-pad],
[left-pad,bottom+pad],[(left+right)/2,bottom+pad],[right+pad,bottom+pad],
[left-pad,(top+bottom)/2],[right+pad,(top+bottom)/2]
];
const vals=[];
for(const [cx,cy] of pts){
const px=Math.max(0,Math.min(pdfCanvas.width-1,Math.round(cx*sx))),py=Math.max(0,Math.min(pdfCanvas.height-1,Math.round(cy*sy)));
const d=pdfCtx.getImageData(px,py,1,1).data;
vals.push([d[0],d[1],d[2]]);
}
const median=i=>vals.map(v=>v[i]).sort((a,b)=>a-b)[Math.floor(vals.length/2)];
const rgb=[median(0),median(1),median(2)];
return '#'+rgb.map(v=>v.toString(16).padStart(2,'0')).join('');
}catch(e){return '#ffffff'}
}
